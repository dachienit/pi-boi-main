FUNCTION zfm_write_po_item.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_EBELN) TYPE  EBELN OPTIONAL
*"----------------------------------------------------------------------

  DATA: lt_ekpo TYPE TABLE OF ekpo,
        ls_ekpo TYPE ekpo.

  SELECT ebeln
         ebelp
         matnr
         menge
         netpr
    FROM ekpo
    INTO TABLE lt_ekpo
   WHERE ebeln = iv_ebeln.

  IF sy-subrc = 0.
    LOOP AT lt_ekpo INTO ls_ekpo.
      WRITE: / ls_ekpo-ebeln,
               ls_ekpo-ebelp,
               ls_ekpo-matnr,
               ls_ekpo-menge,
               ls_ekpo-netpr.
    ENDLOOP.
  ENDIF.

ENDFUNCTION.
